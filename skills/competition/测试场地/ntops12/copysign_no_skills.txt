     4. Pattern for Simple Element-Wise Binary Operators

     The simplest and most canonical pattern is demonstrated by mul (no alpha
     parameter, two inputs, one output):

     Kernel file (src/ntops/kernels/mul.py):
      import functools
      from ninetoothed import Tensor
      from ntops.kernels.element_wise import arrangement

      def application(input, other, output):
          output = input * other  # noqa: F841

      def premake(ndim, dtype=None, block_size=None):
          arrangement_ = functools.partial(arrangement, block_size=block_size)
          tensors = (
              Tensor(ndim, dtype=dtype),
              Tensor(ndim, dtype=dtype),
              Tensor(ndim, dtype=dtype),
          )
          return arrangement_, application, tensors
     Torch wrapper (src/ntops/torch/mul.py):
      import torch
      import ntops
      from ntops.torch.utils import _cached_make

      def mul(input, other, *, out=None):
          if out is None:
              out = torch.empty_like(input)
          kernel = _cached_make(ntops.kernels.mul.premake, input.ndim)
          kernel(input, other, out)
          return out
     Test (tests/test_mul.py):
      @skip_if_cuda_not_available
      @pytest.mark.parametrize(*generate_arguments())
      def test_mul(shape, dtype, device, rtol, atol):
          input = torch.randn(shape, dtype=dtype, device=device)
          other = torch.randn(shape, dtype=dtype, device=device)
          ninetoothed_output = ntops.torch.mul(input, other)
          reference_output = torch.mul(input, other)
          assert torch.allclose(ninetoothed_output,
      reference_output, rtol=rtol, atol=atol)
     Variant with alpha parameter (add and sub): These take an extra alpha
     scalar (as Tensor(0, dtype=ninetoothed.float64)) and compute input +
     alpha * other / input - alpha * other.

     Variant with libdevice (pow): Uses from ninetoothed.language import
     libdevice and calls libdevice.pow(input, exponent) in the application
     function. This is relevant because copysign would likely also use a
     libdevice or ntl function.

     The shared element_wise arrangement (src/ntops/kernels/element_wise.py):
      def arrangement(*tensors, block_size=None):
          if block_size is None:
              block_size = ninetoothed.block_size()
          ndim = max(tensor.ndim for tensor in tensors)
          assert all(tensor.ndim == ndim or tensor.ndim == 0 for tensor
      in tensors)
          return tuple(
              tensor.flatten().tile((block_size,)) if tensor.ndim != 0
      else tensor
              for tensor in tensors
          )
     This flattens all tensors to 1D and tiles them by block_size. Scalars
     (ndim==0) are passed through untouched.

     def application(input, other, output):

             other_sign = other_bits >> 15
